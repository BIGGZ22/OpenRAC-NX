# func_001F6668 (draw.c, FontPrint)

Right size with `char c` (42/640); otherwise 628/636. Written from retail with help from the Lombyte draft.

The function is defined under an asm alias because the file declares FontPrint with `void *` arguments. `color` and `tex` are 64-bit.

Structure:
- `do { c = *p; if ((unsigned)(c - 8) < 8) palette; else if (font[*p].adv) { accent cell; if (*p < 0x20) grey 24-wide; else if (*p > 0x20) normal; x += adv; } } while (*p)`.
- Every font lookup re-reads `*p`.
- The palette update is two statements (`color &= 0xFF000000; color |= ...`).

The key constraint: the loop-top `c = *p` must stay a separate load, neither merged with the condition's load on the back edge nor with later reads. Only a `char c` achieves that (its sign extension differs), but then the load is `lb` where retail has `lbu`. With unsigned types, gcse merges the loads (628). `(unsigned char)c` and `c & 0xFF` fold the loop-top load into the condition's delay slot (636).

```c
struct FontEntry {
    unsigned char u;
    unsigned char v;
    signed char dy;
    signed char adv;
};
extern short D_0015F59C;              /* SDA, gp -0x7764: palette colours on */
extern int D_0015F5A0 MACRO_ADDR;
extern int D_0018CBF8[];

void func_001F6668_d(int x, int y, unsigned long color, unsigned char *str, int count,
                     unsigned long tex, struct FontEntry *font) __asm__("func_001F6668");

/* FontPrint: draw up to count characters of str at (x, y) with the
   16x16 cells of font (u, v, y offset, advance per character). Codes
   8..15 switch to a palette colour (D_0018CBF8) when enabled; 0x80..0xA7
   also draw the cell 0x40 above as an accent; codes below 0x20 are
   24-wide glyphs drawn in the colour's grey. */
void func_001F6668_d(int x, int y, unsigned long color, unsigned char *str, int count,
                     unsigned long tex, struct FontEntry *font) {
    unsigned char *p;
    int i;
    char c;

    if (D_0015F5A0 == 0) {
        D_0018CBF8[0] = color;
    }
    i = 0;
    if (count == 0 || *str == 0) {
        return;
    }
    p = str;
    do {
        c = *p;
        if ((unsigned int)((unsigned char)c - 8) < 8) {
            if (*(int *)&D_0015F59C != 0) {
                color &= 0xFF000000;
                color |= D_0018CBF8[*p - 8] & 0xFFFFFF;
            }
        } else if (font[*p].adv != 0) {
            if ((unsigned char)(*p + 0x80) < 0x28) {
                int cc = *p + 0x40;

                func_001F5800(x + font[cc].adv, y + font[cc].dy, 0x10, 0x10, font[cc].u,
                              font[cc].v, 0x10, 0x10, color, tex);
            }
            if (*p < 0x20) {
                int q = (int)((color & 0xFF) + ((color >> 8) & 0xFF) + ((color >> 16) & 0xFF)) / 3;

                func_001F5800(x, y + font[*p].dy, 0x18, 0x10, font[*p].u, font[*p].v, 0x18,
                              0x10, (int)(color & 0xFF000000) + (q << 16) + (q << 8) + q, tex);
            } else if (*p > 0x20) {
                func_001F5800(x, y + font[*p].dy, 0x10, 0x10, font[*p].u, font[*p].v, 0x10,
                              0x10, color, tex);
            }
            x += font[*p].adv;
        }
        i++;
        if (i == count) {
            break;
        }
        p++;
    } while (*p != 0);
}
```
