# func_00119AA8 (00119868.c, sceTtyWrite, ee29)

340 against retail 336, from the Lombyte draft with PAL names.

The `busy = 0` store goes before the EI call; tools/fix_volatile_slot.py puts it in the slot.

What is left: `pkt` (the third argument of sceDeci2ReqSend) is allocated `$a3` with the 0xA constant in `$a2`, so a `move a2,a3` is added. Retail has pkt in `$a2` and 0xA in `$a3`. The loop-exit path also computes `&state` differently.

```c
struct TtyWriteState {
    int sock;               /* deci2 socket */
    volatile int len;       /* published packet length */
    volatile int unk08;
    volatile int busy;      /* cleared by sceTtyHandler */
    int wbuf;               /* uncached packet header */
    int rbuf;
    int queue;
};
struct TtyWritePacket {
    unsigned short len;
    unsigned short unk02;
    unsigned short unk04;
    unsigned char unk06;
    signed char dest;       /* 0x07 */
    unsigned int unk08;
    unsigned char data[0xF4];
};
extern struct TtyWriteState D_00154B50_w __asm__("D_00154B50");
extern unsigned char D_00154B80_w[] __asm__("D_00154B80");
extern int func_00119768(int, int, void *);   /* sceDeci2ReqSend */
extern int func_00119798(int);                /* sceDeci2Poll */
extern int func_0011D960(void);
extern int func_0011D9A8(void);

/* sceTtyWrite: copy up to 0x100 bytes of buf (a newline sent as CR LF)
   into the uncached DECI2 packet, send it, and poll until the handler
   clears busy. Returns the characters taken, or -1 while a send is in
   flight or when it fails. Adapted from Lombyte (MIT) for PAL. */
int func_00119AA8(char *buf, int len) {
    int done;
    int left;
    int n;
    char *src;
    struct TtyWritePacket *pkt;
    unsigned char *dst;

    done = 0;
    left = len;
    n = 0;
    src = buf;
    if (D_00154B50_w.busy == 0) {
        func_0011D960();
        D_00154B50_w.busy = 1;
        pkt = (struct TtyWritePacket *)((unsigned int)D_00154B80_w | 0x20000000);
        D_00154B50_w.wbuf = (int)pkt;
        dst = pkt->data;
        do {
            left = left - 1;
            if (left == -1) {
                break;
            }
            if (*src == '\n') {
                *dst = '\r';
                n = n + 1;
                dst = dst + 1;
                if (n >= 0x100) {
                    break;
                }
            }
            *dst = (unsigned char)*src;
            n = n + 1;
            src = src + 1;
            dst = dst + 1;
            done = done + 1;
        } while (n < 0x100);
        D_00154B50_w.len = n + 0xC;
        pkt->len = D_00154B50_w.len;
        if (func_00119768(D_00154B50_w.sock, pkt->dest, pkt) < 0) {
            D_00154B50_w.busy = 0;
            func_0011D9A8();
            return -1;
        }
        while (D_00154B50_w.busy != 0) {
            func_00119798(D_00154B50_w.sock);
        }
        func_0011D9A8();
        return done;
    }
    return -1;
}
```
