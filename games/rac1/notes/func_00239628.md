# func_00239628 (vendor.c, 9x9 font text printer)

Same shape as retail; register allocation differs.

Retail spills `x` (sp+0x10, reloaded on newline), `tex` (sp+0x18) and `ch*4` (sp+0x20). It also keeps a constant 1 in s22, re-set each iteration and used for both `blink = 1` (movz) and `hide = 1` (with `slt`/`movz` for `rem < r`). Ours keeps `x` in fp and `ch*4` in s7, so every register after that differs.

/* Prints str with the 9x9 character cells of the 0xE935 font texture,
   scaled by scale, from (x, y): character 0x10A starts a new line at x,
   0x62 makes the next glyph blink, and cells off the left or right edge
   are skipped. D_001E6318 packs each glyph's texture row (>> 20) and
   column (low half, in 16ths); D_001E6418 is its advance. */
void func_00239628(char *str, int x, int y, float scale) {
    long tex;
    int cx;
    int blink = 0;

    tex = func_00200248(func_00200198(0xE935, 0));
    cx = x;
    for (;;) {
        unsigned char c = *str;
        int ch;
        int *ent;
        int hide;

        if (c == 0) {
            return;
        }
        hide = 0;
        ch = c - 0x20;
        str++;
        if (ch == 0x42) {
            blink = 1;
        }
        if (ch == 0xEA) {
            cx = x;
            y += 9;
        }
        if (ch >= 0x3B) {
            continue;
        }
        ent = &D_001E6318[ch];
        if (*ent == -1 && ch != 0) {
            continue;
        }
        if (blink) {
            int period;

            blink = 0;
            period = func_001F98C0(0x1E);
            period += func_001F98C0(10);
            hide = D_0015F538 % period < func_001F98C0(10);
        }
        if (ch != 0 && !hide) {
            int row = (unsigned int)*ent >> 20;
            int col = (*ent & 0xFFFF) >> 4;

            if (cx + 9 > 0 && cx < 0x100) {
                int w = func_001FA898_r(func_001FA888(9) * scale);
                int h = func_001FA898_r(func_001FA888(9) * scale);

                func_001F5800(cx, y, w, h, col, row, 9, 9, 0x80404040L, tex);
            }
        }
        cx += func_001FA898_r(func_001FA888(D_001E6418[ch]) * scale);
    }
}
